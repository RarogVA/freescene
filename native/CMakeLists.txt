cmake_minimum_required(VERSION 3.24)
project(FreeSceneEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(build_tests OFF CACHE BOOL "" FORCE)
set(build_examples OFF CACHE BOOL "" FORCE)
set(python-bindings OFF CACHE BOOL "" FORCE)
set(webtorrent OFF CACHE BOOL "" FORCE)
set(OPENSSL_USE_STATIC_LIBS TRUE)
if(ANDROID)
    list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/android")
    add_library(nlohmann_json INTERFACE)
    target_include_directories(nlohmann_json INTERFACE "${FREESCENE_JSON_INCLUDE}")
    add_library(nlohmann_json::nlohmann_json ALIAS nlohmann_json)
else()
    find_package(nlohmann_json 3.12 REQUIRED)
endif()
FetchContent_Declare(libtorrent GIT_REPOSITORY https://github.com/arvidn/libtorrent.git GIT_TAG 6da363d2994f17c0b3c0450d124cf73a31a73847 GIT_SHALLOW FALSE GIT_SUBMODULES deps/try_signal)
FetchContent_MakeAvailable(libtorrent)
add_library(FreeSceneEngine SHARED engine.cpp)
target_link_libraries(FreeSceneEngine PRIVATE torrent-rasterbar nlohmann_json::nlohmann_json)
set_target_properties(FreeSceneEngine PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(ANDROID)
    target_sources(FreeSceneEngine PRIVATE android/jni.cpp)
    target_link_options(FreeSceneEngine PRIVATE "-Wl,-z,max-page-size=16384")
else()
    set_target_properties(FreeSceneEngine PROPERTIES INSTALL_NAME_DIR "@rpath")
endif()
